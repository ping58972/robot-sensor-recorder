cmake_minimum_required(VERSION 3.20)
project(robot_sensor_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(recorder src/recording.cpp)
target_include_directories(recorder PUBLIC include)
target_link_libraries(recorder PUBLIC Threads::Threads)
target_compile_options(recorder PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sensor-recorder src/main.cpp src/sources.cpp)
target_link_libraries(sensor-recorder PRIVATE recorder)
target_compile_options(sensor-recorder PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(queue-check tests/queue_check.cpp)
    target_link_libraries(queue-check PRIVATE recorder)
    target_compile_options(queue-check PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME queue_lifecycle COMMAND queue-check)
    add_test(NAME recorder_integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:sensor-recorder>)
    set_tests_properties(queue_lifecycle recorder_integration PROPERTIES TIMEOUT 30)
endif()
